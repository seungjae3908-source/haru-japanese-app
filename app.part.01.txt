  function render(){
    setTheme();
    let content='';
    if(state.tab==='today') content=viewToday();
    if(state.tab==='course') content=viewCourse();
    if(state.tab==='pronunciation') content=viewPronunciation();
    if(state.tab==='review') content=viewReview();
    if(state.tab==='profile') content=viewProfile();
    $app.innerHTML=appShell(content);
    if(state.tab==='pronunciation') drawEmptyWave();
  }
