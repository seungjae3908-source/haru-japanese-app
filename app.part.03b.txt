  function resetRecorderUi(clear=true){
    if(clear){audioUrl='';recordTranscript='';const p=document.getElementById('audioPlayer');if(p){p.classList.add('hidden');p.removeAttribute('src')}}
    const timer=document.getElementById('recordTimer');if(timer)timer.textContent='00:00';
    const res=document.getElementById('pronResult');if(clear&&res)res.innerHTML='<div class="card empty"><span class="emoji">🎧</span>녹음을 완료하면 분석 결과가 여기에 표시됩니다.</div>';
    drawEmptyWave(); setRecordingUi(false);
  }

  function startLiveWave(stream){
    const C=window.AudioContext||window.webkitAudioContext;
    liveAudioContext=new C(); liveAnalyser=liveAudioContext.createAnalyser(); liveAnalyser.fftSize=1024;
    liveAudioContext.createMediaStreamSource(stream).connect(liveAnalyser); liveData=new Uint8Array(liveAnalyser.fftSize);
    drawLiveWave();
  }
  function drawLiveWave(){
    const canvas=document.getElementById('wave');if(!canvas||!liveAnalyser)return;
    const ctx=canvas.getContext('2d');liveAnalyser.getByteTimeDomainData(liveData);
    ctx.clearRect(0,0,canvas.width,canvas.height);ctx.strokeStyle=getComputedStyle(document.documentElement).getPropertyValue('--primary');ctx.lineWidth=3;ctx.beginPath();
    const slice=canvas.width/liveData.length;let x=0;
    liveData.forEach((v,i)=>{const y=v/128*canvas.height/2;i?ctx.lineTo(x,y):ctx.moveTo(x,y);x+=slice});ctx.stroke();
    waveformRaf=requestAnimationFrame(drawLiveWave);
  }
  function drawEmptyWave(){
    const canvas=document.getElementById('wave');if(!canvas)return;const ctx=canvas.getContext('2d');ctx.clearRect(0,0,canvas.width,canvas.height);ctx.strokeStyle=getComputedStyle(document.documentElement).getPropertyValue('--line');ctx.lineWidth=2;ctx.beginPath();ctx.moveTo(0,canvas.height/2);ctx.lineTo(canvas.width,canvas.height/2);ctx.stroke();
  }
  function stopLiveAudio(){cancelAnimationFrame(waveformRaf);waveformRaf=0;if(liveAudioContext){liveAudioContext.close().catch(()=>{});liveAudioContext=null}liveAnalyser=null;liveData=null;}

  async function finishPronunciation(blob,lang){
    const l=current(), target=langText(l,lang), duration=Math.max(.1,(Date.now()-recordStartedAt)/1000);
    let audioMetrics={silenceRatio:.5,pitchRange:0,amplitudeVariation:0,duration};
    try{audioMetrics=await analyzeAudio(blob)}catch(e){console.warn('audio analysis',e)}
    const accuracy=recordTranscript?similarity(recordTranscript,target,lang):null;
    const speed=clamp(Math.round(100-Math.abs(Math.log(audioMetrics.duration/expectedDuration(target,lang)))*72),20,100);
    const fluency=clamp(Math.round(100-Math.max(0,audioMetrics.silenceRatio-.08)*170),20,100);
    const intonation=audioMetrics.pitchRange?clamp(Math.round(55+Math.min(audioMetrics.pitchRange,8)*5-Math.max(audioMetrics.pitchRange-12,0)*3),30,100):50;
    const accuracyWeight=accuracy==null?0:.58;
    const denom=accuracyWeight+.14+.16+.12;
    const overall=Math.round(((accuracy||0)*accuracyWeight+speed*.14+fluency*.16+intonation*.12)/denom);
    const feedback=[];
    if(accuracy==null)feedback.push({type:'warn',text:'이 브라우저에서 음성 인식 결과를 받지 못해 문장 정확도는 점수에 포함하지 않았습니다.'});
    else if(accuracy>=88)feedback.push({type:'good',text:'목표 문장이 매우 정확하게 인식됐습니다.'});
    else if(accuracy>=65)feedback.push({type:'warn',text:'일부 단어·음절이 다르게 인식됐습니다. 표시된 구간을 천천히 반복하세요.'});
    else feedback.push({type:'bad',text:'빠지거나 바뀐 구간이 많습니다. 느린 음성을 듣고 짧게 나누어 말해 보세요.'});
    if(speed<65)feedback.push({type:'warn',text:audioMetrics.duration>expectedDuration(target,lang)?'목표보다 느립니다. 문장 덩어리를 연결해 보세요.':'목표보다 빠릅니다. 음절과 끝소리를 생략하지 마세요.'});else feedback.push({type:'good',text:'문장 길이에 맞는 말하기 속도입니다.'});
    if(fluency<65)feedback.push({type:'warn',text:'침묵이나 끊김이 많습니다. 의미 단위로 한 번에 말하는 연습이 필요합니다.'});
    if(intonation<60)feedback.push({type:'warn',text:'높낮이 변화가 적습니다. 핵심 단어와 문장 끝 억양을 살려 보세요.'});
    feedback.push({type:'good',text:`우선 교정 포인트: ${l.tips[lang]}`});
    const diffHtml=recordTranscript?buildDiffHtml(recordTranscript,target,lang):'';
    const result={lessonId:l.id,time:Date.now(),lang,transcript:recordTranscript,target,accuracy,speed,fluency,intonation,overall,duration:audioMetrics.duration,silenceRatio:audioMetrics.silenceRatio,pitchRange:audioMetrics.pitchRange,feedback,diffHtml};
    state.pronunciationHistory.push(result);state.scores.push({id:l.id,score:overall,time:Date.now(),type:'pronunciation'});
    if((accuracy??100)<80)state.mistakes.push({id:l.id,type:'pronunciation',target,input:recordTranscript,score:accuracy||0,time:Date.now()});
    else addXp(12,'발음 훈련 완료');
    save();
    const box=document.getElementById('pronResult');if(box)box.innerHTML=pronunciationResultHtml(result);
    setRecordingUi(false,false);const retry=document.getElementById('retryRecord');if(retry)retry.disabled=false;
  }

  async function analyzeAudio(blob){
    const buffer=await blob.arrayBuffer();const C=window.AudioContext||window.webkitAudioContext;const ctx=new C();
    const audio=await ctx.decodeAudioData(buffer.slice(0));const channel=audio.getChannelData(0);const sr=audio.sampleRate;const frame=Math.max(512,Math.floor(sr*.04));
    const rms=[],pitches=[];
