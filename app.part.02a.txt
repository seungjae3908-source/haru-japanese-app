  function pronunciationResultHtml(r){
    const score = r.overall ?? 0;
    return `<div class="card">
      <div class="section-title"><div><h2>발음 분석 결과</h2><p>${new Date(r.time).toLocaleString('ko-KR')}</p></div></div>
      <div class="grid">
        <div class="center"><div class="score-ring" style="--score:${score}"><strong>${score}</strong></div><b class="mt8">종합 연습 점수</b><div class="muted">정밀 음소 진단이 아닌 웹 1차 분석</div></div>
        <div>
          ${metric('문장 인식 정확도',r.accuracy,r.transcript?'말한 내용이 목표 문장과 얼마나 일치하는지':'음성 인식을 사용할 수 없어 미측정')}
          ${metric('말하기 속도',r.speed,'목표 문장 길이에 맞는 속도')}
          ${metric('유창성',r.fluency,'긴 침묵과 끊김의 비율')}
          ${metric('억양 변화',r.intonation,'목소리 높낮이가 지나치게 평평하지 않은지')}
        </div>
      </div>
      <div class="divider"></div>
      <b>인식된 문장</b><div class="feedback ${r.accuracy>=75?'good':'warn'} mt8">${esc(r.transcript || '이 브라우저에서 음성 인식 결과를 받지 못했습니다.')}</div>
      ${r.diffHtml || ''}
      <div class="feedback-list">
        ${r.feedback.map(f=>`<div class="feedback ${f.type}">${esc(f.text)}</div>`).join('')}
      </div>
    </div>`;
  }
  function metric(name,value,desc){
    return `<div class="metric"><div class="metric-head"><b>${name}</b><span>${value==null?'—':value+'점'}</span></div><div class="progress"><span style="width:${value||0}%"></span></div><small class="muted">${desc}</small></div>`;
  }
