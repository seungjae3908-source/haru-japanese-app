  function checkRecall(){
    const input=document.getElementById('recallInput')?.value||'';
    const l=current(), {target}=pathParts();
    const targetText=langText(l,target), score=similarity(input,targetText,target);
    const ok=score>=80;
    const result=document.getElementById('answerResult');
    result.innerHTML=`<div class="feedback ${ok?'good':'bad'} mt12"><b>${score}점</b><br>${ok?'잘 기억했습니다.':'정답: '+esc(targetText)}</div>`;
    state.scores.push({id:l.id,score,time:Date.now(),type:'recall'});
    if(!ok) state.mistakes.push({id:l.id,type:'recall',target:targetText,input,score,time:Date.now()});
    else addXp(8,'회상 성공');
    save();
  }

  function markLearned(){
    const l=current();
    if(!state.learned.some(v=>v.id===l.id&&v.date===todayKey())) state.learned.push({id:l.id,date:todayKey(),time:Date.now()});
    addXp(5,'오늘 문장 학습');
    render();
  }

  function saveWord(index){
    const l=current(),w=l.vocab[index];
    const item={ko:w[0],ja:w[1],en:w[2],lessonId:l.id};
    if(!state.savedWords.some(v=>v.ko===item.ko&&v.lessonId===item.lessonId)){
      state.savedWords.push(item); addXp(2,'단어 저장'); save();
    }else toast('이미 저장한 단어입니다.');
  }
