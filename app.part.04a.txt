  }
  function autoCorrelate(buf,sr){
    let rms=0;for(const v of buf)rms+=v*v;rms=Math.sqrt(rms/buf.length);if(rms<.01)return-1;
    const min=Math.floor(sr/450),max=Math.min(Math.floor(sr/70),buf.length-1);let best=-1,bestCorr=0;
    for(let lag=min;lag<=max;lag+=2){let corr=0;for(let i=0;i<buf.length-lag;i++)corr+=buf[i]*buf[i+lag];if(corr>bestCorr){bestCorr=corr;best=lag}}
    return best>0?sr/best:-1;
  }

  function buildDiffHtml(input,target,lang){
    const expected=lang==='en'?target.toLowerCase().replace(/[^a-z0-9' ]/g,'').split(/\s+/):lang==='ko'?target.replace(/[\p{P}\p{S}]/gu,'').split(/\s+/):Array.from(target.replace(/[\s\p{P}\p{S}]/gu,''));
    const heard=lang==='en'?input.toLowerCase().replace(/[^a-z0-9' ]/g,'').split(/\s+/):lang==='ko'?input.replace(/[\p{P}\p{S}]/gu,'').split(/\s+/):Array.from(input.replace(/[\s\p{P}\p{S}]/gu,''));
    const used=new Set();
    const tokens=expected.map(e=>{let best=-1,bestScore=0;heard.forEach((h,i)=>{if(used.has(i))return;const s=similarity(h,e,lang);if(s>bestScore){bestScore=s;best=i}});if(best>=0&&bestScore>=65){used.add(best);return`<span class="token ok">${esc(e)}</span>`}return`<span class="token miss">${esc(e)}</span>`});
    heard.forEach((h,i)=>{if(!used.has(i))tokens.push(`<span class="token extra">+ ${esc(h)}</span>`)});
    return `<div class="mt12"><b>구간 비교</b><div class="diff">${tokens.join('')}</div><small class="muted">초록: 인식됨 · 빨강: 빠짐/다름 · 주황: 추가로 인식됨</small></div>`;
  }
