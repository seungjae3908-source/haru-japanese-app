  function load(){
    try{
      const saved = JSON.parse(localStorage.getItem(KEY) || 'null');
      return saved ? {...structuredClone(defaultState), ...saved, reminders:{...defaultState.reminders,...(saved.reminders||{})}} : structuredClone(defaultState);

  function clamp(n,min,max){ return Math.min(max,Math.max(min,n)); }

  function normalize(text, lang){
    let s = String(text || '').toLowerCase().normalize('NFKC');
    s = s.replace(/[\s\p{P}\p{S}]/gu,'');
    if(lang === 'en') s = s.replace(/[^a-z0-9]/g,'');
    return s;
  }
  function levenshtein(a,b){
    const m=a.length,n=b.length,dp=Array(n+1).fill(0).map((_,i)=>i);
    for(let i=1;i<=m;i++){
      let prev=dp[0]; dp[0]=i;
      for(let j=1;j<=n;j++){
        const old=dp[j];
        dp[j]=Math.min(dp[j]+1,dp[j-1]+1,prev+(a[i-1]===b[j-1]?0:1));
        prev=old;
      }
    }
    return dp[n];
  }
  function similarity(input,target,lang){
    const a=normalize(input,lang), b=normalize(target,lang);
    if(!a || !b) return 0;
    return Math.round((1-levenshtein(a,b)/Math.max(a.length,b.length))*100);
  }
  function expectedDuration(text,lang){
    if(lang==='en') return Math.max(1.4, text.trim().split(/\s+/).length * .48 + .35);
    return Math.max(1.4, normalize(text,lang).length * .19 + .45);
  }

  function speak(text, lang, rate=1){
    if(!('speechSynthesis' in window)){ toast('이 브라우저는 음성 재생을 지원하지 않습니다.'); return; }
    speechSynthesis.cancel();
    const u = new SpeechSynthesisUtterance(text);
    u.lang = LANG[lang].locale; u.rate = rate; u.pitch = 1;
    const voices = speechSynthesis.getVoices();
    const voice = voices.find(v=>v.lang.toLowerCase().startsWith(lang==='ja'?'ja':lang==='ko'?'ko':'en'));
    if(voice) u.voice = voice;
    speechSynthesis.speak(u);
  }
